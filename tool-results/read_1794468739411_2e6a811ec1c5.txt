   124→class TocDocTemplate(SimpleDocTemplate):
   125→    def afterFlowable(self, flowable):
   126→        if hasattr(flowable, "bookmark_name"):
   127→            level = getattr(flowable, "bookmark_level", 0)
   128→            text = getattr(flowable, "bookmark_text", "")
   129→            key = getattr(flowable, "bookmark_key", "")
   130→            self.notify("TOCEntry", (level, text, self.page, key))

   133→def on_page(canvas, doc):
   134→    canvas.saveState()
   135→    # header: doc title + accent rule
   136→    canvas.setFont("FreeSerif", 7.5)
   137→    canvas.setFillColor(TEXT_MUTED)
   138→    canvas.drawString(MARGIN, PAGE_H - 0.55 * inch, DOC_TITLE)
   139→    canvas.setStrokeColor(ACCENT)
   140→    canvas.setLineWidth(1.2)
   141→    canvas.line(MARGIN, PAGE_H - 0.62 * inch, PAGE_W - MARGIN, PAGE_H - 0.62 * inch)
   142→    # footer: author + page number (TOC page = i, body arabic resets to 1)
   143→    canvas.setStrokeColor(BORDER)
   144→    canvas.setLineWidth(0.6)
   145→    canvas.line(MARGIN, 0.62 * inch, PAGE_W - MARGIN, 0.62 * inch)
   146→    canvas.setFont("FreeSerif", 7.5)
   147→    canvas.setFillColor(TEXT_MUTED)
   148→    canvas.drawString(MARGIN, 0.45 * inch, "AppMOT · MVP v1.0 · ноябрь 2026")
   149→    if doc.page == 1:
   150→        num = "i"
   151→    else:
   152→        num = str(doc.page - 1)
   153→    canvas.drawRightString(PAGE_W - MARGIN, 0.45 * inch, num)
   154→    canvas.restoreState()

   162→def add_heading(text, style, level=0):
   163→    key = "h_" + hashlib.md5(text.encode()).hexdigest()[:8]
   164→    p = Paragraph(f'<a name="{key}"/><b>{text}</b>', style)
   165→    p.bookmark_name = key
   166→    p.bookmark_level = level
   167→    p.bookmark_text = text
   168→    p.bookmark_key = key
   169→    return p

   194→def safe_keep(elements):
   195→    total = 0
   196→    for el in elements:
   197→        w, h = el.wrap(AVAIL_W, PAGE_H)
   198→        total += h
   199→    if total <= MAX_KEEP_HEIGHT:

   207→def make_table(head, rows, ratios, header_align="C", repeat=True):
   208→    assert abs(sum(ratios) - 1.0) < 0.01, "ratios must sum to 1"
   209→    col_w = [r * AVAIL_W for r in ratios]
   210→    assert sum(col_w) <= AVAIL_W + 0.5, "table wider than available width"
   211→    ha = S["cellHC"] if header_align == "C" else S["cellH"]
   212→    data = [[Paragraph(f"<b>{h}</b>", ha) for h in head]]
   213→    for row in rows:
   214→        data.append([c if isinstance(c, (Paragraph, Image)) else Paragraph(str(c), S["cell"]) for c in row])
   215→    t = Table(data, colWidths=col_w, hAlign="CENTER", repeatRows=1 if repeat else 0)
   216→    style = [
   217→        ("BACKGROUND", (0, 0), (-1, 0), TABLE_HEADER_COLOR),
   218→        ("VALIGN", (0, 0), (-1, -1), "MIDDLE"),
   219→        ("GRID", (0, 0), (-1, -1), 0.5, BORDER),
   220→        ("LEFTPADDING", (0, 0), (-1, -1), 7),
   221→        ("RIGHTPADDING", (0, 0), (-1, -1), 7),
   222→        ("TOPPADDING", (0, 0), (-1, -1), 5.5),
   223→        ("BOTTOMPADDING", (0, 0), (-1, -1), 5.5),
   224→    ]
   225→    for i in range(1, len(data)):
   226→        style.append(("BACKGROUND", (0, i), (-1, i), TABLE_ROW_EVEN if i % 2 == 1 else TABLE_ROW_ODD))
   227→    t.setStyle(TableStyle(style))
   228→    return t

   247→def fit_image(path, max_w, max_h):
   248→    im = PILImage.open(path)
   249→    ow, oh = im.size
   250→    rw = max_w / ow if ow > max_w else 1.0
   251→    rh = max_h / oh if oh > max_h else 1.0
   252→    r = min(rw, rh)
   253→    return Image(path, width=ow * r, height=oh * r)
